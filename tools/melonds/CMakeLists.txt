cmake_minimum_required(VERSION 3.16)
project(gbar3_melonds_probe LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(BUILD_QT_SDL OFF CACHE BOOL "" FORCE)
set(ENABLE_JIT OFF CACHE BOOL "" FORCE)
set(ENABLE_GDBSTUB OFF CACHE BOOL "" FORCE)
set(ENABLE_OGLRENDERER OFF CACHE BOOL "" FORCE)
set(ENABLE_LTO_RELEASE OFF CACHE BOOL "" FORCE)
if(NOT MELONDS_SOURCE)
  message(FATAL_ERROR "Set MELONDS_SOURCE to melonDS cce3e49252392801da758ad41ed66d5f730d9a07 with cache-lab.patch applied (see README.md)")
endif()
add_subdirectory("${MELONDS_SOURCE}" melonDS)
add_executable(gbar3-melonds-probe main.cpp platform.cpp)
target_link_libraries(gbar3-melonds-probe PRIVATE core)
if(MINGW)
  target_link_options(gbar3-melonds-probe PRIVATE -static -static-libgcc -static-libstdc++)
endif()
